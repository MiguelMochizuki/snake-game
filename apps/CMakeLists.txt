	add_executable(snake-gui snake_gui.cpp)
	target_link_libraries(snake-gui PRIVATE snake_gui_core)
	snake_game_set_warnings(snake-gui)
	if(WIN32)
		# Windows: embed the icon and run as a GUI program (no console window), keeping plain main()
		target_sources(snake-gui PRIVATE ${CMAKE_SOURCE_DIR}/assets/snake-gui.rc)
		set_target_properties(snake-gui PROPERTIES WIN32_EXECUTABLE TRUE)
		if(MSVC)
			target_link_options(snake-gui PRIVATE /ENTRY:mainCRTStartup)
		endif()
	endif()
